rules_version = '2';

// Règles de sécurité Firestore du site Altus (à coller dans la console Firebase
// du projet Altus : Firestore Database > Règles > remplacer tout > Publier).
//  - Le groupe (ALTUS) n'existe que si un COACH l'a ouvert.
//  - Un joueur connecté (compte Google) peut lire les bilans
//    du groupe et écrire SEULEMENT son propre bilan.
//  - Défi de la semaine et exos donnés : écrits par les coachs ; un joueur peut
//    seulement noter la date où il a fait ses exos (champ « faits »).
// ⚠ Ce fichier est public : avant de coller, remplacer ADRESSE_COACH_1, 2…
//   par les adresses Gmail des coachs (ne pas les écrire ici).
service cloud.firestore {
  match /databases/{database}/documents {

    function estCoach() {
      return request.auth != null
        && request.auth.token.email in ['ADRESSE_COACH_1', 'ADRESSE_COACH_2']
        && request.auth.token.email_verified == true;
    }
    function connecte() { return request.auth != null; }
    function groupeExiste(code) { return exists(/databases/$(database)/documents/groupes/$(code)); }

    match /groupes/{code} {
      allow get: if connecte();            // vérifier qu'un code existe
      allow list: if estCoach();
      allow create, update, delete: if estCoach() && code.matches('[A-Z0-9-]{3,20}');

      function bilanValide() {
        let d = request.resource.data;
        return d.keys().hasOnly(['uid', 'prenom', 'maj', 'annee', 'exos', 'hist', 'semaine', 'defi', 'source', 'repris', 'dist'])
          && d.prenom is string && d.prenom.size() > 0 && d.prenom.size() <= 30
          && d.exos is map && d.exos.size() <= 60
          && d.hist is map && d.hist.size() <= 60
          && (!('dist' in d) || (d.dist is map && d.dist.size() <= 20));
      }

      match /bilans/{uid} {
        allow read: if connecte() && groupeExiste(code);
        // Un joueur écrit son bilan ; un coach peut écrire tous les bilans (imports du classeur, correction d'un score)
        allow create, update: if groupeExiste(code) && bilanValide()
          && ((connecte() && uid == request.auth.uid && request.resource.data.uid == request.auth.uid)
              || (estCoach() && request.resource.data.uid == uid));
        allow delete: if estCoach() || (connecte() && uid == request.auth.uid);
      }

      // Match play : un joueur entre un match dont il est le joueur 1 ; personne ne le modifie ;
      // l'auteur ou un coach peut l'effacer
      match /matchs/{id} {
        allow read: if connecte() && groupeExiste(code);
        allow create: if connecte() && groupeExiste(code)
          && request.resource.data.keys().hasOnly(['auteur', 'j1', 'j2', 'gagnant', 'score', 'date', 'parcours', 'cree'])
          && request.resource.data.auteur == request.auth.uid
          && request.resource.data.j1 is string && request.resource.data.j1.size() > 0 && request.resource.data.j1.size() <= 30
          && request.resource.data.j2 is string && request.resource.data.j2.size() > 0 && request.resource.data.j2.size() <= 30
          && request.resource.data.gagnant in [0, 1, 2]
          && request.resource.data.score is string && request.resource.data.score.size() <= 10
          && request.resource.data.date is string && request.resource.data.date.size() == 10
          && request.resource.data.parcours is string && request.resource.data.parcours.size() <= 60;
        allow delete: if estCoach() || (connecte() && resource.data.auteur == request.auth.uid);
      }

      // Sauvegarde complète des scores d'un joueur : lui seul l'écrit ; lui et les coachs la lisent (rapport Excel)
      match /sauvegardes/{uid} {
        allow read: if estCoach() || (connecte() && uid == request.auth.uid && groupeExiste(code));
        allow write: if estCoach() || (connecte() && uid == request.auth.uid && groupeExiste(code));
      }

      match /reglages/{id} {
        allow read: if connecte() && groupeExiste(code);
        allow write: if estCoach();
      }

      match /consignes/{uid} {
        allow read: if estCoach() || (connecte() && uid == request.auth.uid);
        allow create, delete: if estCoach();
        allow update: if estCoach()
          || (connecte() && uid == request.auth.uid
              && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['faits', 'ressentis'])
              && request.resource.data.faits is map && request.resource.data.faits.size() <= 10
              && (!('ressentis' in request.resource.data) || (request.resource.data.ressentis is list && request.resource.data.ressentis.size() <= 30)));
      }
    }
  }
}
